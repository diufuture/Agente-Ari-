// ───────────────────────── PUNTO DE ENTRADA ─────────────────────────

function doPost(e) {
  var candado = LockService.getScriptLock();
  try {
    candado.waitLock(30000);
  } catch (errorCandado) {
    return respuestaJSON({ ok: false, motivo: 'El servidor está ocupado, intenta de nuevo en unos segundos.' });
  }

  try {
    if (!e || !e.postData || !e.postData.contents) {
      throw new Error('No llegaron datos en la solicitud.');
    }

    var datos = JSON.parse(e.postData.contents);

    guardarFila(datos);

    // Si un correo falla, no debe tumbar el guardado ya hecho.
    try { enviarCorreoAdmin(datos); } catch (errorCorreoAdmin) { /* seguimos */ }
    try { enviarCorreoCliente(datos); } catch (errorCorreoCliente) { /* seguimos */ }

    return respuestaJSON({ ok: true });
  } catch (error) {
    return respuestaJSON({ ok: false, motivo: String(error && error.message ? error.message : error) });
  } finally {
    candado.releaseLock();
  }
}

// ───────────────────────── GUARDAR EN LA HOJA ─────────────────────────

function obtenerOCrearHojaRespuestas() {
  var libro = SpreadsheetApp.getActiveSpreadsheet();
  var hoja = libro.getSheetByName(NOMBRE_HOJA_RESPUESTAS);
  if (!hoja) {
    hoja = libro.insertSheet(NOMBRE_HOJA_RESPUESTAS);
    hoja.getRange(1, 1, 1, ENCABEZADOS_RESPUESTAS.length).setValues([ENCABEZADOS_RESPUESTAS]);
    hoja.setFrozenRows(1);
  }
  return hoja;
}

function guardarFila(datos) {
  var hoja = obtenerOCrearHojaRespuestas();

  var ultimaColumna = Math.max(hoja.getLastColumn(), 1);
  var encabezados = hoja.getRange(1, 1, 1, ultimaColumna).getDisplayValues()[0];

  var fila = encabezados.map(function (encabezado) {
    var clave = String(encabezado || '').trim();
    if (/fecha/i.test(clave) && /(registro|env[ií]o|timestamp)/i.test(clave)) {
      return Utilities.formatDate(new Date(), 'America/Bogota', 'yyyy-MM-dd HH:mm:ss');
    }
    var valor = datos[clave];
    return (valor === undefined || valor === null) ? '' : String(valor);
  });

  var siguienteFila = hoja.getLastRow() + 1;
  var rango = hoja.getRange(siguienteFila, 1, 1, fila.length);
  rango.setNumberFormat('@'); // texto plano: que no reformatee teléfonos, aptos, fechas
  rango.setValues([fila]);
}

function promedioCalificacion(datos) {
  var campos = ['Cal Claridad', 'Cal Dominio', 'Cal Atencion', 'Cal General'];
  var valores = campos.map(function (campo) { return Number(datos[campo]) || 0; });
  var suma = valores.reduce(function (a, b) { return a + b; }, 0);
  return Math.round((suma / campos.length) * 10) / 10;
}

// ───────────────────────── CORREO AL CLIENTE ─────────────────────────

function enviarCorreoCliente(datos) {
  var correo = datos['Correo'];
  if (!correo || String(correo).indexOf('@') === -1) return;

  var nombreCompleto = String(datos['Nombre'] || '').trim();
  var primerNombre = nombreCompleto.split(' ')[0] || 'vecino';
  var promedio = promedioCalificacion(datos);

  MailApp.sendEmail({
    to: correo,
    subject: '¡Gracias por confiar en Click Control! · Amazonía 96',
    htmlBody: plantillaCorreoCliente(primerNombre, datos, promedio),
    name: 'Click Control'
  });
}

function estrellasHTML(valor) {
  var n = Math.max(0, Math.min(5, Math.round(Number(valor) || 0)));
  var html = '';
  for (var i = 0; i < 5; i++) {
    html += '<span style="color:' + (i < n ? '#F5C21B' : '#3A4A5A') + ';font-size:20px;">★</span>';
  }
  return html;
}

function respuestaJSON(objeto) {
